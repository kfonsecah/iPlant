import { PlantIdentificationResult, PlantaInterface } from "../types-dtos/plant.types";

// Plant.id API configuration
const PLANT_ID_API_URL = "https://api.plant.id/v3/identification";
// TODO: Replace with actual API key from environment or config
const PLANT_ID_API_KEY = process.env.EXPO_PUBLIC_PLANT_ID_API_KEY || "";

/**
 * Convert image URI to base64 data URI format required by Plant.id
 */
async function imageToDataUri(uri: string): Promise<string> {
  const FileSystem = require('expo-file-system');
  const base64 = await FileSystem.readAsStringAsync(uri, {
    encoding: FileSystem.EncodingType.Base64,
  });
  // Detect format from URI or default to jpeg
  const isPng = uri.toLowerCase().includes('.png') || uri.includes('png');
  const mimeType = isPng ? 'image/png' : 'image/jpeg';
  return `data:${mimeType};base64,${base64}`;
}

/**
 * Identify plant using Plant.id API
 * @param imageUri - Local file URI of plant photo
 * @returns PlantIdentificationResult with name, confidence, description, care
 */
export async function identifyPlant(imageUri: string): Promise<PlantIdentificationResult> {
  if (!PLANT_ID_API_KEY) {
    throw new Error("PLANT_ID_API_KEY not configured");
  }

  const dataUri = await imageToDataUri(imageUri);

  const response = await fetch(PLANT_ID_API_URL, {
    method: "POST",
    headers: {
      "Api-Key": PLANT_ID_API_KEY,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      images: [dataUri],
      latitude: 49.207, // Default location
      longitude: 16.608,
      similar_images: true,
    }),
  });

  if (!response.ok) {
    const errorText = await response.text();
    if (response.status === 429) {
      throw new Error("Rate limit exceeded. Try again later.");
    }
    throw new Error(`Plant.id API error: ${response.status} - ${errorText}`);
  }

  const data = await response.json();

  // Extract first (best) suggestion
  const suggestion = data.suggestions?.[0];
  if (!suggestion) {
    throw new Error("No plant identified. Try with a clearer photo.");
  }

  return {
    plantName: suggestion.plant_name || "Unknown plant",
    latinName: suggestion.plant_details?.wiki_name,
    probability: Math.round((suggestion.probability || 0) * 100),
    description: suggestion.description,
    careInstructions: suggestion.wiki_description?.extract,
    wikiDescription: suggestion.wiki_description
      ? {
          title: suggestion.wiki_description.title,
          extract: suggestion.wiki_description.extract,
        }
      : undefined,
  };
}
