**2. ConfidenceBadge.tsx**
```typescript
import React from "react";
import { View, Text, StyleSheet } from "react-native";
import { useConfidenceBadgeStyles, CONFIDENCE_COLORS, CONFIDENCE_LABELS } from "./ConfidenceBadge.styles";

interface ConfidenceBadgeProps {
  confidence: number; // 0-100 percentage
  showLabel?: boolean;
  size?: "small" | "medium" | "large";
}

export default function ConfidenceBadge({ 
  confidence, 
  showLabel = true,
  size = "medium" 
}: ConfidenceBadgeProps) {
  const styles = useConfidenceBadgeStyles();
  
  // Determine level and color
  const getConfidenceLevel = (value: number) => {
    if (value >= 70) return "high";
    if (value >= 40) return "medium";
    return "low";
  };
  
  const level = getConfidenceLevel(confidence);
  const color = CONFIDENCE_COLORS[level];
  const label = CONFIDENCE_LABELS[level];
  
  // Size adjustments
  const sizeStyles = {
    small: { container: { paddingHorizontal: 8, paddingVertical: 4 }, text: 12, dot: 6 },
    medium: { container: {}, text: 14, dot: 8 },
    large: { container: { paddingHorizontal: 16, paddingVertical: 8 }, text: 16, dot: 10 },
  }[size];
  
  return (
    <View 
      style={[
        styles.container, 
        sizeStyles.container,
        { backgroundColor: color + "20" } // 20% opacity background
      ]}
    >
      <View style={[styles.dot, { backgroundColor: color, width: sizeStyles.dot, height: sizeStyles.dot }]} />
      <Text style={[styles.text, { color, fontSize: sizeStyles.text }]}>
        {confidence}%{showLabel ? ` · ${label}` : ""}
      </Text>
    </View>
  );
}
```
